## 🔧 技术实现

### 写作助手

```javascript
// 写作模板
const templates = {
  moments: {
    name: '朋友圈文案',
    prompt: '帮我写一条朋友圈，主题是：{theme}，风格：{style}'
  },
  xiaohongshu: {
    name: '小红书文案',
    prompt: '帮我写一篇小红书笔记，主题：{theme}，包含：标题、正文、标签'
  },
  email: {
    name: '工作邮件',
    prompt: '帮我写一封邮件，收件人：{recipient}，主题：{subject}，内容：{content}'
  }
};

// 调用AI生成
async function generateWriting(type, params) {
  const template = templates[type];
  const prompt = template.prompt.replace(/{(\w+)}/g, (match, key) => params[key]);
  
  const response = await fetch('/api/chat', {
    method: 'POST',
    body: JSON.stringify({
      model: 'glm-4-flash',
      messages: [{role: 'user', content: prompt}]
    })
  });
  
  return response.json();
}
```
